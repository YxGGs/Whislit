-- Refresh character on respawn

    if newState == Enum.HumanoidStateType.Freefall then
        freefallStartTime = tick()
    end
end)

-- Utility: blacklist all player characters
local function getPlayerBlacklist()
    local list = {}
    for _, plr in pairs(Players:GetPlayers()) do
        if plr.Character then
            table.insert(list, plr.Character)
        end
    end
    return list
end

-- Enhanced Raycast for walls
local function detectWall(dir, offset, distance)
    local origin = root.Position + (offset or Vector3.new(0, 0, 0))
    local dirVec = dir * (distance or 0.01)
    local params = RaycastParams.new()
    params.FilterDescendantsInstances = getPlayerBlacklist()
    params.FilterType = Enum.RaycastFilterType.Blacklist
    return workspace:Raycast(origin, dirVec, params)
end

-- Multi-direction wall normal detection
local function getWallNormal()
    local dirs = {
        root.CFrame.LookVector,
        root.CFrame.RightVector,
        -root.CFrame.RightVector,
        root.CFrame.LookVector + root.CFrame.RightVector,
        root.CFrame.LookVector - root.CFrame.RightVector,
        -root.CFrame.LookVector,
    }
    for _, d in ipairs(dirs) do
        local res = detectWall(d, nil, 5)
        if res then
            return res.Normal, d
        end
    end
    return nil, nil
end

-- Detect platform covered by a ceiling
local function detectCoveredPlatform()
    local params = RaycastParams.new()
    params.FilterDescendantsInstances = getPlayerBlacklist()
    params.FilterType = Enum.RaycastFilterType.Blacklist

    local floorOrigin = root.Position + Vector3.new(0, 1, 0)
    local floorRes = workspace:Raycast(floorOrigin, Vector3.new(0, -5, 0), params)
    if floorRes then
        local coverRes = workspace:Raycast(floorRes.Position, Vector3.new(0, 1, 0), params)
        if coverRes then
            return true, floorRes, coverRes
        end
    end
    return false
end

-- Modified Wallhop rotation & jump (removed lock to wall)
local function vertTurn()
    local normal, hitDir = getWallNormal()
    if normal then
        local orig = -normal
        local angle = math.acos(normal:Dot(-hitDir)) - math.pi/2
        -- Just apply the rotation without locking to wall
        root.CFrame = root.CFrame * CFrame.Angles(0, angle, 0)
        task.wait(0.075)
    end
end

local function vertJump()
    if human:GetState() == Enum.HumanoidStateType.Freefall then
        isWallhopping = true
        lastHopTime = tick()
        vertTurn()
        human.Jump = true
        hopTimer = tick() + 0.1
        task.wait(0.075)
    end
end

-- Main Loop: detect & execute wallhop
RunService.RenderStepped:Connect(function()
    if tick() < hopTimer then return end

    if isWallhopping then
        if human:GetState() ~= Enum.HumanoidStateType.Freefall or (tick() - lastHopTime > 0.7) then
            isWallhopping = false
        end
    end

    local front = detectWall(root.CFrame.LookVector, Vector3.new(0, 1, 0), 2)
    local fallSpeed = root.Velocity.Y
    if vertActive and front and front.Distance < 3 and fallSpeed < -4 and (tick() - freefallStartTime > 0.2) then
        vertJump()
    end

    local covered, floorRes, coverRes = detectCoveredPlatform()
    if vertActive and covered then
        vertTurn()
        if human:GetState() == Enum.HumanoidStateType.Freefall then
            human.Jump = true
            hopTimer = tick() + 0.01
        end
    end
end)

-- Toggle Movement Flick
vertBtn.MouseButton1Click:Connect(function()
    vertActive = not vertActive
    if vertActive then
        vertBtn.Text = "Movement flick: ON"
        TweenService:Create(vertBtn, TweenInfo.new(0.3), {BackgroundColor3 = Color3.new(0, 1, 0)}):Play()
    else
        vertBtn.Text = "Movement flick: OFF"
        TweenService:Create(vertBtn, TweenInfo.new(0.3), {BackgroundColor3 = Color3.new(0.2, 0.2, 0.2)}):Play()
    end
end)
